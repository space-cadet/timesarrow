% A chain site has four solid magnetic legs and a separate dashed physical index.
\newcommand{\chainsite}[3]{%
 \node[tensor,minimum size=8.5mm] (v#1) at (#2,#3) {$I_{#1}$};
 \draw[magnetic] (v#1.north) -- ++(0,.82) node[above=2pt,tinylabel] {$u_{#1}$};
 \draw[magnetic] (v#1.south) -- ++(0,-.82) node[below=2pt,tinylabel] {$d_{#1}$};
 \draw[physical] (v#1.45) -- ++(.48,.48) node[above right=-1pt,tinylabel,text=violet] {$s_{#1}$};
}
\newcommand{\minipair}[3]{%
 \node[tensor,minimum size=7.8mm] (#1L) at ($(#2)+(-1.55,0)$) {$I^s$};
 \node[tensor,minimum size=7.8mm] (#1R) at ($(#2)+(1.55,0)$) {$I^t$};
 \node[#3,minimum width=11mm,minimum height=8mm] (#1B) at (#2) {};
 \draw[magnetic] (#1L.east) -- (#1B.west);
 \draw[magnetic] (#1B.east) -- (#1R.west);
 \foreach \ang in {145,180,215}{\draw[magnetic] (#1L.\ang) -- ++(\ang:.5);}
 \foreach \ang in {35,0,-35}{\draw[magnetic] (#1R.\ang) -- ++(\ang:.5);}
}
