% Chain site: 4 solid magnetic legs (left/right + u_i/d_i) and angled dashed physical leg s_i.
\newcommand{\chainsite}[3]{%
  \coordinate (c#1) at (#2,#3);
  \begin{scope}[on background layer]
    \draw[magnetic] (c#1) -- ++(0, 1.18) node[above=2pt, leglabel] {$u_{#1}$};
    \draw[magnetic] (c#1) -- ++(0,-1.18) node[below=2pt, leglabel] {$d_{#1}$};
    \draw[physical] (c#1) -- ++(-46:0.96) node[physdot] {} node[below right=-1pt, physlabel] {$s_{#1}$};
  \end{scope}
  \node[tensor, minimum size=8.8mm] (v#1) at (c#1) {$I_{#1}$};
}

% Compact two-node pair for Figure 03 with consistent physical legs s, t.
\newcommand{\tripletpair}[3]{%
  \coordinate (#1Lpos) at ($(#2)+(-1.42,0)$);
  \coordinate (#1Rpos) at ($(#2)+( 1.42,0)$);
  \begin{scope}[on background layer]
    \draw[magnetic] (#1Lpos) -- (#1Rpos);
    \foreach \ang in {142,180,218}{%
      \draw[magnetic] (#1Lpos) -- ++(\ang:0.95);
    }
    \foreach \ang in {38,0,-38}{%
      \draw[magnetic] (#1Rpos) -- ++(\ang:0.95);
    }
    \draw[physical] (#1Lpos) -- ++(0,0.92) node[physdot] {} node[above=2pt, physlabel] {$s$};
    \draw[physical] (#1Rpos) -- ++(0,0.92) node[physdot] {} node[above=2pt, physlabel] {$t$};
  \end{scope}
  \node[tensor, minimum size=8.2mm] (#1L) at (#1Lpos) {$I_A$};
  \node[tensor, minimum size=8.2mm] (#1R) at (#1Rpos) {$I_B$};
  \node[triplet, minimum width=11.4mm, minimum height=7.6mm] (#1B) at (#2) {$#3$};
}
